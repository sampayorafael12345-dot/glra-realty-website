# Robots file for GLRA Realty
# The public site is open to every search engine.

User-agent: *
Allow: /

# The API answers JSON, never a page worth indexing.
Disallow: /api/

# The two sign-in portals are NOT listed here on purpose. Both send
# "X-Robots-Tag: noindex, nofollow" and carry a noindex meta tag, which is the
# reliable way to keep a page out of the results: a crawler has to be allowed
# in to read "do not index me". Disallowing them here would instead have
# advertised where they are - robots.txt is the first file any scanner asks
# for - while leaving them eligible to appear as a bare URL.

Sitemap: https://glrarealty.com/sitemap.xml
